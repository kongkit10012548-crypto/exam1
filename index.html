// ห้องสอบ · Edge Function บน Supabase // รับคำสั่งชุดเดียวกับ Code.gs เพื่อให้ app.js ใช้ได้โดยแก้แค่ API_URL // เพิ่มระบบห้องรอ: join (เข้าห้องรอ) และ setLive (ครูกดเริ่มสอบพร้อมกัน) import { createClient } from "npm:@supabase/supabase-js@2.45.4"; const sb = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!, { auth: { persistSession: false } }, ); const TEACHER_TOKEN = Deno.env.get("TEACHER_TOKEN") ?? ""; const GRACE_MS = 60_000; const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "content-type, authorization, apikey, x-client-info", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const reply = (obj: unknown) => new Response(JSON.stringify(obj), { headers: { ...CORS, "Content-Type": "application/json" } }); const fail = (error: string) => ({ ok: false, error }); Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response(null, { headers: CORS }); if (req.method !== "POST") return reply(fail("method")); let b: any; try { b = await req.json(); } catch { return reply(fail("bad_json")); } try { switch (b.action) { case "exams": return reply({ ok: true, exams: await listExams() }); case "status": return reply({ ok: true, attempts: await myAttempts(b) }); case "join": return reply(await joinAttempt(b)); case "start": return reply(await startAttempt(b)); case "save": return reply(await saveAttempt(b, false)); case "submit": return reply(await saveAttempt(b, true)); case "teacher": return reply(await teacherOp(b)); default: return reply(fail("unknown_action")); } } catch (e) { return reply(fail(String((e as Error).message || e))); } }); /* ---------- ตัวช่วย ---------- */ const normName = (s: unknown) => String(s ?? ""); return /^[A-Za-z0-9_-]{1,60}$/.test(s) ? s : ""; } function cleanAnswers(ans: any, questions: any[]): Record { const out: Record = {}; if (!ans || typeof ans !== "object") return out; for (const q of questions || []) { const v = ans[q.id]; if (typeof v === "number" && Number.isInteger(v) && v >= 0 && v < (q.choices || []).length) out[q.id] = v; } return out; } function cleanEvents(evs: any): any[] { if (!Array.isArray(evs)) return []; const out: any[] = []; for (const e of evs.slice(-300)) { if (!e || typeof e.t !== "string" || typeof e.at !== "number") continue; const c: any = { t: e.t.slice(0, 20), at: e.at }; if (e.r != null) c.r = String(e.r).slice(0, 10); if (e.d != null) c.d = Number(e.d) || 0; out.push(c); } return out; } /* ---------- แปลงแถวจากฐานข้อมูลเป็นรูปแบบที่หน้าเว็บใช้ ---------- */ const toExam = (r: any) => ({ id: r.id, title: r.title, minutes: r.minutes, proctor: r.proctor, shuffle: r.shuffle, open: r.open, questions: r.questions, updatedAt: Number(r.updated_at), liveAt: Number(r.live_at || 0), }); const toAtt = (r: any) => ({ examId: r.exam_id, title: r.title, no: r.no, name: r.name, proctor: r.proctor, startedAt: Number(r.started_at), endAt: Number(r.end_at), status: r.status, submittedAt: r.submitted_at == null ? undefined : Number(r.submitted_at), answers: r.answers || {}, events: r.events || [], updatedAt: Number(r.updated_at), }); /* ---------- ข้อมูล ---------- */ async function listExams() { const { data, error } = await sb.from("exams").select("*").order("updated_at", { ascending: false }); if (error) throw error; return (data ?? []) { if (normName(r.name) === who.name) out[r.exam_id] = toAtt(r); } return out; } /* ---------- นักเรียน ---------- */ // เข้าห้องรอ: บันทึกชื่อไว้ก่อน ยังไม่จับเวลา async function joinAttempt(b: any) { const who = checkStudent(b); if ("error" in who) return fail(who.error); const examId = cleanId(b.examId); const ex = examId ? await getExam(examId) : null; if (!ex) return fail("no_exam"); const a = await getAttempt(examId, who.no); if (a && normName(a.name) !== who.name) return fail("name_mismatch"); if (a && a.status === "submitted") return { ok: false, error: "submitted", attempt: a }; if (a) return { ok: true, attempt: a }; if (!ex.open) return fail("closed"); const now = Date.now(); const waiting = { examId, title: ex.title, no: who.no, name: who.name, proctor: !!ex.proctor, startedAt: 0, endAt: 0, status: "waiting", answers: {}, events: [], updatedAt: now, }; await writeAtt(waiting); return { ok: true, attempt: waiting }; } // เริ่มทำข้อสอบ: ถ้าอยู่ในห้องรอ จะเริ่มนับเวลาจากเวลาที่ครูกดเริ่ม async function startAttempt(b: any) { const who = checkStudent(b); if ("error" in who) return fail(who.error); const examId = cleanId(b.examId); const ex = examId ? await getExam(examId) : null; if (!ex) return fail("no_exam"); const a = await getAttempt(examId, who.no); if (a && normName(a.name) !== who.name) return fail("name_mismatch"); if (a && a.status === "submitted") return { ok: false, error: "submitted", attempt: a }; const now = Date.now(); // เคยเริ่มแล้ว (รีเฟรช/เปิดใหม่) ให้ทำต่อจากเดิม if (a && a.status !== "waiting") { if (now > a.endAt + GRACE_MS) return finalize(a, "timeout", now); if (!ex.open) return fail("closed"); a.events.push({ t: "rejoin", at: now }); a.status = "doing"; a.updatedAt = now; await writeAtt(a); return { ok: true, attempt: a }; } // เริ่มพร้อมกัน: ทุกคนใช้เวลาเริ่มและเวลาหมดเดียวกันตามที่ครูกด if (!ex.liveAt) return fail("not_live"); const startedAt = ex.liveAt; const endAt = startedAt + (Number(ex.minutes) || 30) * 60000; if (now > endAt + GRACE_MS) return fail("ended"); const att: any = a || { examId, title: ex.title, no: who.no, name: who.name, proctor: !!ex.proctor, answers: {}, events: [], }; att.title = ex.title; att.proctor = !!ex.proctor; att.startedAt = startedAt; att.endAt = endAt; att.status = "doing"; att.events.push({ t: "start", at: now }); att.updatedAt = now; await writeAtt(att); return { ok: true, attempt: att }; } async function saveAttempt(b: any, isSubmit: boolean) { const who = checkStudent(b); if ("error" in who) return fail(who.error); const examId = cleanId(b.examId); const ex = examId ? await getExam(examId) : null; if (!ex) return fail("no_exam"); const a = await getAttempt(examId, who.no); if (!a) return fail("reset"); if (normName(a.name) !== who.name) return fail("name_mismatch"); if (a.status === "submitted") return { ok: true, attempt: a }; if (a.status === "waiting") return fail("not_live"); const now = Date.now(); a.answers = cleanAnswers(b.answers, ex.questions); a.events = cleanEvents(b.events); if (isSubmit) return finalize(a, b.kind === "timeout" ? "timeout" : "submit", now); if (now > a.endAt + GRACE_MS) return finalize(a, "timeout", now); a.status = b.status === "away" ? "away" : "doing"; a.updatedAt = now; await writeAtt(a); return { ok: true, attempt: a }; } async function finalize(a: any, kind: string, now: number) { const last = a.events[a.events.length - 1]; if (!last || (last.t !== "submit" && last.t !== "timeout")) a.events.push({ t: kind, at: now }); a.status = "submitted"; a.submittedAt = now; a.updatedAt = now; await writeAtt(a); return { ok: true, attempt: a }; } /* ---------- ผู้สอน (ต้องมีรหัส) ---------- */ async function teacherOp(b: any) { if (!TEACHER_TOKEN || String(b.token ?? "") !== TEACHER_TOKEN) return fail("unauthorized"); switch (b.op) { case "list": { const exams = await listExams(); const { data: kd, error: ke } = await sb.from("exam_keys").select("*"); if (ke) throw ke; const keys: Record = {}; for (const k of kd ?? []) keys[k.exam_id] = { answers: k.answers, updatedAt: Number(k.updated_at) }; const { data: ad, error: ae } = await sb.from("attempts").select("*"); if (ae) throw ae; return { ok: true, exams, keys, attempts: (ad ?? []).map(toAtt) }; } case "saveExam": return saveExam(b); case "deleteExam": { const id = cleanId(b.examId); if (!id) return fail("bad_id"); const { error } = await sb.from("exams").delete().eq("id", id); if (error) throw error; return { ok: true }; } case "toggleOpen": { const id = cleanId(b.examId); if (!id) return fail("bad_id"); const { error } = await sb.from("exams").update({ open: !!b.open, updated_at: Date.now() }).eq("id", id); if (error) throw error; return { ok: true }; } case "setLive": { // เปิด = เริ่มสอบพร้อมกันทันที (และเปิดรับด้วย) / ปิด = ยกเลิกการเริ่ม const id = cleanId(b.examId); if (!id) return fail("bad_id"); const now = Date.now(); const on = !!b.live; const patch: any = { live_at: on ? now : 0, updated_at: now }; if (on) patch.open = true; const { error } = await sb.from("exams").update(patch).eq("id", id); if (error) throw error; return { ok: true }; } case "reset": { const id = cleanId(b.examId); const no = String(b.no ?? "").trim(); if (!id || !no) return fail("bad_request"); const { error } = await sb.from("attempts").delete().eq("exam_id", id).eq("no", no); if (error) throw error; return { ok: true }; } default: return fail("unknown_op"); } } async function saveExam(b: any) { const e = b.exam || {}; const id = cleanId(e.id); if (!id) return fail("bad_id"); const title = String(e.title ?? "").trim(); if (!title) return fail("no_title"); const minutes = Math.round(Number(e.minutes)); if (!(minutes >= 1 && minutes <= 600)) return fail("bad_minutes"); const questions: any[] = []; const src: any[] = e.questions || []; for (let i = 0; i < src.length; i++) { const q = src[i]; const choices = (q.choices || []).map((c: any) => String(c ?? "").trim()).filter(Boolean); if (!String(q.text ??
